 36def class_to_num(s):
 37    """Return the numeric limit of a weight class."""
 38    m = re.match(r"\d+(\.\d+)?", str(s))
 39    return float(m.group()) if m else np.nan

 47class Cleaner:
 48    """Builds engineered variables from explanatory columns."""
 49    def fit(self, d):
 50        """Learn the category frequency tables."""
 51        self.counts_ = {c: d[c].astype(str).value_counts() for c in COUNT_COLS}
 52        return self
 53    def transform(self, r):
 54        """Return a copy of a dataset with engineered variables."""
 55        d = r.copy()
 56        for c in CAT:
 57            d[c] = d[c].astype(str)
 58        age, bw = d["Age"], d["BodyweightKg"]
 59        wc = d["WeightClassKg"].map(class_to_num)
 60        d["IsMale"] = (d["Sex"] == "M").astype(int)
 61        d["WeightClass_num"] = wc
 62        d["WeightClassPlus"] = d["WeightClassKg"].str.endswith("+").astype(int)
 63        d["BW_minus_Class"] = bw - wc
 64        d["BirthYear"] = d["MeetYear"] - age
 65        d["IsJunior"] = (age < 23).astype(int)
 66        d["IsMaster"] = (age >= 40).astype(int)
 67        d["IsRaw"] = (d["Equipment"] == "Raw").astype(int)
 68        d["IsTested"] = (d["Tested"] == "Yes").astype(int)
 69        d["IsUSA"] = (d["Country"] == "USA").astype(int)
 70        d["SameCountry"] = ((d["Country"] == d["MeetCountry"]) & (d["Country"] != "Unknown")).astype(int)
 71        d["ParentFedUnknown"] = (d["ParentFederation"] == "Unknown").astype(int)
 72        d["StateUnknown"] = (d["State"] == "Unknown").astype(int)
 73        d["Age_log"] = np.log(age.clip(lower=0.1))
 74        d["BodyweightKg_log"] = np.log(bw.clip(lower=1))
 75        d["BW_x_Male"] = bw * d["IsMale"]
 76        for c in COUNT_COLS:
 77            d[f"{c}_count"] = d[c].map(self.counts_[c]).fillna(0)
 78        return d.replace([np.inf, -np.inf], np.nan)

 80class Preprocessor:
 81    """Cleaning and encoding fitted: produces a "scl" view (imputed, scaled, one-hot)
 82    and an "ord" view (unscaled numerics with NaN, then ordinal-coded categoricals)."""
 83    def __init__(self, num=NUM_VARS, cat=CAT_VARS):
 84        self.num, self.cat = list(num), list(cat)
 85    def _frame(self, d):
 86        """Clean a dataset, keep the model variables and cast categoricals to str."""
 87        d = self.cleaner.transform(d)[self.num + self.cat].copy()
 88        d[self.cat] = d[self.cat].astype(str)
 89        return d
 90    def fit(self, d):
 91        """Fit the cleaner and the encoders."""
 92        self.cleaner = Cleaner().fit(d)
 93        f = self._frame(d)
 94        self.ct = ColumnTransformer([
 95            ("num", Pipeline([("imp", SimpleImputer(strategy="median")), ("sc", StandardScaler())]), self.num),
 96            ("cat", OneHotEncoder(handle_unknown="infrequent_if_exist", min_frequency=0.005, sparse_output=False),
 97             self.cat)]).fit(f)
 98        self.oe = OrdinalEncoder(handle_unknown="use_encoded_value", unknown_value=np.nan).fit(f[self.cat])
 99        return self
100    @property
101    def ord_cat_idx(self):
102        """Categorical column positions in the "ord" view."""
103        return list(range(len(self.num), len(self.num) + len(self.cat)))
104    def _scl(self, f):
105        """Encode a cleaned frame as the "scl" view."""
106        return np.clip(self.ct.transform(f), -10, 10).astype(np.float32)
107    def _ord(self, f):
108        """Encode a cleaned frame as the "ord" view."""
109        num = f[self.num].apply(pd.to_numeric, errors="coerce").to_numpy(np.float32)
110        return np.hstack([num, self.oe.transform(f[self.cat]).astype(np.float32)])
111    def transform(self, d):
112        """Return the "scl" view."""
113        return self._scl(self._frame(d))
114    def transform_views(self, d):
115        """Return {"scl": array, "ord": array}."""
116        f = self._frame(d)
117        return {"scl": self._scl(f), "ord": self._ord(f)}

121class SeedBag:
122    """Average of HistGradientBoosting model."""
123    def __init__(self, params, n_seeds=3):
124        self.m = [HistGradientBoostingRegressor(**params, random_state=SEED + s) for s in range(n_seeds)]

126        """Fit every model."""
127        for m in self.m:

131        """Return the mean prediction."""
132        return np.mean([m.predict(X) for m in self.m], axis=0)

134class EarlyStopped:
135    """Adds early stopping to a booster using an internal 10% validation split."""
136    def __init__(self, model, fit_kw=lambda: {}):
137        self.model, self.fit_kw = model, fit_kw

139        """Fit with the inner split as eval set."""

147class TorchMLP:
148    """Multi-layer perceptron of Linear-BatchNorm-Sigmoid-Dropout blocks; Adam with early stopping, best epoch restored."""
149    def __init__(self, hidden=(128, 64), dropout=0.1, lr=1e-3, batch=512, max_epochs=100, patience=10,
150                 val_frac=0.1, seed=SEED):
151        self.hidden, self.dropout, self.lr, self.batch = hidden, dropout, lr, batch
152        self.max_epochs, self.patience, self.val_frac, self.seed = max_epochs, patience, val_frac, seed
153    def _net(self, d):
154        """Build the network for `d` input features."""
155        layers, prev = [], d
156        for h in self.hidden:
157            layers += [nn.Linear(prev, h), nn.BatchNorm1d(h), nn.Sigmoid(), nn.Dropout(self.dropout)]
158            prev = h
159        return nn.Sequential(*layers, nn.Linear(prev, 1))
160    def fit(self, X, y):
161        """Train with early stopping on an inner validation split."""
162        torch.manual_seed(self.seed)
163        self.mu_, self.sd_ = float(np.mean(y)), float(np.std(y))
164        z = (np.asarray(y) - self.mu_) / self.sd_
165        Xa, Xb, za, zb = train_test_split(X, z, test_size=self.val_frac, random_state=self.seed)
166        t = lambda a: torch.tensor(a, dtype=torch.float32, device=DEVICE)
167        Xa, Xb, za, zb = t(Xa), t(Xb), t(za), t(zb)
168        self.net = self._net(X.shape[1]).to(DEVICE)
169        opt, loss_fn = torch.optim.Adam(self.net.parameters(), lr=self.lr), nn.MSELoss()
170        best, self.best_epoch_, best_state = np.inf, 0, None
171        bar = tqdm(range(1, self.max_epochs + 1), unit="epoch")
172        for ep in bar:
173            self.net.train()
174            perm, total = torch.randperm(len(Xa), device=DEVICE), 0.0
175            for i in range(0, len(Xa), self.batch):
176                idx = perm[i:i + self.batch]
177                if len(idx) < 2:
178                    continue
179                opt.zero_grad()
180                loss = loss_fn(self.net(Xa[idx]).squeeze(1), za[idx])
181                loss.backward()
182                opt.step()
183                total += loss.item() * len(idx)
184            self.net.eval()
185            with torch.no_grad():
186                val_loss = loss_fn(self.net(Xb).squeeze(1), zb).item()
187            if val_loss < best - 1e-5:
188                best, self.best_epoch_ = val_loss, ep
189                best_state = {k: v.clone() for k, v in self.net.state_dict().items()}
190            bar.set_description(f"Epoch {ep}/{self.max_epochs}")
191            bar.set_postfix(loss=f"{total / len(Xa):.4f}", val_loss=f"{val_loss:.4f}", best_epoch=self.best_epoch_)
192            if ep - self.best_epoch_ >= self.patience:
193                break
194        bar.close()
195        self.net.load_state_dict(best_state)
196        return self
197    def predict(self, X):
198        """Return predictions on the original target scale."""
199        self.net.eval()
200        with torch.no_grad():
201            z = self.net(torch.tensor(X, dtype=torch.float32, device=DEVICE)).squeeze(1).cpu().numpy()
202        return z * self.sd_ + self.mu_

206class InContextModel:
207    """Base class for in-context models that predict without training."""
208    view = "ord"
209    def __init__(self, cat_idx=None, max_context=50_000, batch_rows=10_000, seed=SEED):
210        self.cat_idx, self.max_context, self.batch_rows, self.seed = list(cat_idx or []), max_context, batch_rows, seed
211        self.m = None
212    def fit(self, X, y):
213        """Store the context and load the model."""
214        X, y = np.asarray(X, np.float32), np.asarray(y, np.float64)
215        if self.max_context and len(X) > self.max_context:
216            X, _, y, _ = train_test_split(X, y, train_size=self.max_context, stratify=target_bins(y),
217                                          random_state=self.seed)
218        self.X_, self.y_ = X, y
219        self._load()
220        return self
221    def predict(self, X):
222        """Return predictions, scored in batches."""
223        X = np.asarray(X, np.float32)
224        return np.concatenate([np.asarray(self._predict(X[i:i + self.batch_rows]), dtype=np.float64)
225                               for i in range(0, len(X), self.batch_rows)])
226    def release(self):
227        """Free the model and GPU memory."""
228        self.m = self.xc = self.yc = None
229        gc.collect()
230        if torch.cuda.is_available():
231            torch.cuda.empty_cache()
232    def _frame(self, X):
233        """Return a DataFrame with categorical columns typed as 'category'."""
234        df = pd.DataFrame(X, columns=[f"f{i}" for i in range(X.shape[1])])
235        for i in self.cat_idx:
236            df[f"f{i}"] = df[f"f{i}"].astype("category")
237        return df
238    def _load(self):
239        """Build the model."""
240        raise NotImplementedError
241    def _predict(self, X):
242        """Return predictions for a batch."""
243        raise NotImplementedError

245class TabPFNModel(InContextModel):
246    """TabPFN, a pretrained transformer for tabular regression."""
247    def __init__(self, version="3.5", size="small", n_estimators=None, **kw):
248        super().__init__(**kw)
249        self.version, self.size, self.n_estimators = version, size, n_estimators
250    def _load(self):
251        """Create and fit the TabPFN regressor."""
252        from tabpfn import TabPFNRegressor
253        from tabpfn.constants import ModelVersion
254        v = {"2.6": ModelVersion.V2_6,
255             "3.5": ModelVersion.V3_5_FAST if self.size == "small" else ModelVersion.V3_5}[self.version]
256        opts = dict(device=DEVICE, categorical_features_indices=self.cat_idx or None,
257                    ignore_pretraining_limits=True, random_state=self.seed)
258        if self.n_estimators:
259            opts["n_estimators"] = self.n_estimators
260        self.m = TabPFNRegressor.create_default_for_version(v, **opts)
261        self.m.fit(self.X_, self.y_)
262    def _predict(self, X):
263        """Return the mean of the predictive distribution."""
264        return self.m.predict(X)

281class KumoTabularModel(InContextModel):
282    """Kumo-Tabular, an in-context model for tabular classification."""
283    def __init__(self, size="small", **kw):
284        super().__init__(**kw)
285        self.size = size
286    def _table(self, X, y=None):
287        """Convert arrays to an sdm TableTensor, with an optional target column."""
288        import sdm
289        df = pd.DataFrame(X, columns=[f"f{i}" for i in range(X.shape[1])])
290        stypes = {c: "numerical" for c in df.columns}
291        if y is not None:
292            df["target"], stypes["target"] = y, "numerical"
293        return sdm.TableTensor.from_pandas(df=df, stypes=stypes, device=torch.device(DEVICE))
294    def _load(self):
295        """Create the model and build the context tables."""
296        import sdm
297        self.m = sdm.models.KumoTabular(task="regression", size=self.size, device=DEVICE)
298        ctx = self._table(self.X_, self.y_)
299        self.xc, self.yc = ctx.drop_columns("target"), ctx[:, "target"]
300    def _predict(self, X):
301        """Return the mean over the predicted quantiles."""
302        dev = torch.device(DEVICE)
303        with torch.amp.autocast(dev.type, dtype=torch.float16, enabled=dev.type == "cuda"):
304            out = self.m(x_context=self.xc, y_context=self.yc, x_query=self._table(X), num_estimators=16)
305        return out.to_pandas().to_numpy(dtype=np.float64).mean(axis=1)

309def take(X, idx):
310    """Select rows from an array or a dict of views."""
311    return {k: v[idx] for k, v in X.items()} if isinstance(X, dict) else X[idx]

313def stack_views(A, B):
314    """Concatenate rows of two arrays or two dicts of views."""
315    return {k: np.vstack([A[k], B[k]]) for k in A} if isinstance(A, dict) else np.vstack([A, B])

317def _view(model, X):
318    """Select the feature view the model requires (default "scl")."""
319    return X[getattr(model, "view", "scl")] if isinstance(X, dict) else X

321def fit_predict(model, X_train, y_train, X_eval):
322    """Fit *model* and return predictions for *X_eval*."""
323    p = np.asarray(model.fit(_view(model, X_train), y_train).predict(_view(model, X_eval)), dtype=np.float64)
324    if hasattr(model, "release"):
325        model.release()
326    return p

328def run_model(name, model, models, P, X_train, y_train, X_eval, raise_errors=False):
329    """Fit *model*, storing it in *models* and its predictions in *P*."""
330    try:
331        P[name] = fit_predict(model, X_train, y_train, X_eval)
332        models[name] = model
333    except Exception as e:
334        if raise_errors:
335            raise
336        print(f"[{name}] skipped: {type(e).__name__}: {e}")

338def report(y_true, p):
339    """Return RMSE, MAE, median absolute error, R2 and MAPE."""
340    return {"RMSE": mean_squared_error(y_true, p) ** 0.5,
341            "MAE": mean_absolute_error(y_true, p),
342            "MedAE": median_absolute_error(y_true, p),
343            "R2": r2_score(y_true, p),
344            "MAPE %": 100 * np.mean(np.abs(y_true - p) / np.abs(y_true))}
345    
346def results_table(P, y_true):
347    """Return the metrics table sorted by RMSE."""
348    return pd.DataFrame({n: report(y_true, p) for n, p in P.items()}).T.sort_values("RMSE").round(4)

350def refit_predict(models, best, X, y, X_new):
351    """Refit the selected model on (X, y) and predict *X_new*."""
352    return fit_predict(models[best], X, y, X_new)

150    def __init__(self, hidden=(128, 64), dropout=0.1, lr=1e-3, batch=512, max_epochs=100, patience=10,
151                 val_frac=0.1, seed=SEED):
152        self.hidden, self.dropout, self.lr, self.batch = hidden, dropout, lr, batch
153        self.max_epochs, self.patience, self.val_frac, self.seed = max_epochs, patience, val_frac, seed
154    def _net(self, d):
155        """Build the network for `d` input features."""
156        layers, prev = [], d
157        for h in self.hidden:
158            layers += [nn.Linear(prev, h), nn.BatchNorm1d(h), nn.Sigmoid(), nn.Dropout(self.dropout)]
159            prev = h
160        return nn.Sequential(*layers, nn.Linear(prev, 1))
161    def fit(self, X, y):
162        """Train with early stopping on an inner validation split."""
163        torch.manual_seed(self.seed)
164        self.mu_, self.sd_ = float(np.mean(y)), float(np.std(y))
165        z = (np.asarray(y) - self.mu_) / self.sd_
166        Xa, Xb, za, zb = train_test_split(X, z, test_size=self.val_frac, random_state=self.seed)
167        t = lambda a: torch.tensor(a, dtype=torch.float32, device=DEVICE)
168        Xa, Xb, za, zb = t(Xa), t(Xb), t(za), t(zb)
169        self.net = self._net(X.shape[1]).to(DEVICE)
170        opt, loss_fn = torch.optim.Adam(self.net.parameters(), lr=self.lr), nn.MSELoss()
171        best, self.best_epoch_, best_state = np.inf, 0, None
172        bar = tqdm(range(1, self.max_epochs + 1), unit="epoch")
173        for ep in bar:
174            self.net.train()
175            perm, total = torch.randperm(len(Xa), device=DEVICE), 0.0

177                idx = perm[i:i + self.batch]
178                if len(idx) < 2:

181                loss = loss_fn(self.net(Xa[idx]).squeeze(1), za[idx])
182                loss.backward()

192            bar.set_postfix(loss=f"{total / len(Xa):.4f}", val_loss=f"{val_loss:.4f}", best_epoch=self.best_epoch_)
193            if ep - self.best_epoch_ >= self.patience:
